var agt = navigator.userAgent.toLowerCase();
var is_op = (agt.indexOf("opera") != -1);
var is_ie = (agt.indexOf("msie") != -1) && document.all && !is_op;
var is_mac = (agt.indexOf("mac") != -1);
var is_gk = (agt.indexOf("gecko") != -1);
var is_sf = (agt.indexOf("safari") != -1) || (agt.indexOf("applewebkit") != -1);

function isBrowserCompatible() {
	if (is_ie && !is_op && !is_mac) {
		var v = gff(agt, "msie ");
		if (v != null) {
			return (v >= 6.0);
		}
	}
	if (is_gk && !is_sf) {
		var v = gff(agt, "rv:");
		if (v != null) {
			return (v >= 1.4);
		} else {
			v = gff(agt, "galeon/");
			if (v != null) {
				return (v >= 1.3);
			}
		}
	}
	if(is_op){
		var v = gff(agt, "opera/");
		if (v != null) {
			return (v >= 4.0);
		}
	}
	if (is_sf) {
		var v = gff(agt, "applewebkit/");
		if (v != null) {
			return (v >= 124);
		}
	}
	return false;
}
function gff(str, pfx) {
	var i = str.indexOf(pfx);
	if (i != -1) {
		var v = parseFloat(str.substring(i + pfx.length));
		if (!isNaN(v)) {
			return v;
		}
	}
	return null;
}

function strNullCheck(str){
	if(str == null) str = '';
	if(str == '' || str == ' '){
		return '';
	} else if(str.length){
		var tempChar = '';
		var spaceCharCount = 0;
		for(var i = 0; i < str.length; i++){
			tempChar = str.charAt(i);
			if(tempChar == ' '){
				spaceCharCount++;				
			}
		}
		if(spaceCharCount == str.length){
			return '';
		} else {
			return str;
		}
	}
}

//This function can be used for validating two dates and checking if the first date is before the second date. The objects should be passed and not the value
function compareDates(startDate,endDate){
	var sDate="";
	var eDate="";
	if(startDate && startDate.value.trim() !=""){
		sDate=parseDate(startDate.value);
		if(sDate==""){alert("Please enter a valid Start Date. "); return false;}
	}else{
		alert("Please enter a valid Start Date. "); return false;
	}
	if(endDate && endDate.value.trim()!=""){
		eDate=parseDate(endDate.value);
		if(eDate==""){alert("Please enter a valid End Date. "); return false;}
	}else{
		alert("Please enter a valid End Date. "); return false;
	}	
	if(startDate && endDate && sDate !="" && eDate !=""){
		startDate.value=sDate;
		endDate.value=eDate;
		if(Date.parse(sDate) > Date.parse(eDate)){
			alert("Start Date cannot be after End Date.");
			return false;
		}
	}
	return true;
}

//return the date in MM/dd/yyyy format
function parseDate(date) {
	var month = "";
	var day = "";
	var year = "";

	if (date.indexOf("-") > -1) {
		// Dates with -'s in it
		var splitDate = date.split("-");
		if (splitDate.length == 3) {
			month = splitDate[0];
			day = splitDate[1];
			year = splitDate[2];
			if(month.length>2 || day.length>2 || year.length>4)
				return "";
		} else {
			return "";
		}
	} else if (date.indexOf("/") > -1) {
		// Dates with /'s in it
		var splitDate = date.split("/");
		if (splitDate.length == 3) {
			month = splitDate[0];
			day = splitDate[1];
			year = splitDate[2];
			if(month.length>2 || day.length>2 || year.length>4)
				return "";
		} else {
			return "";
		}
	} else {
		// All other dates
		if (date.length == 5) {
			date = "0" + date;
			month = date.substring(0, 2);
			day = date.substring(2, 4);
			year = date.substring(4);
		} else if (date.length == 6) {
			month = date.substring(0, 2);
			day = date.substring(2, 4);
			year = date.substring(4);
		} else if (date.length == 7) {
			date = "0" + date;
			month = date.substring(0, 2);
			day = date.substring(2, 4);
			year = date.substring(4);
		} else if (date.length == 8) {
			month = date.substring(0, 2);
			day = date.substring(2, 4);
			year = date.substring(4);
		} else {
			return "";
		}
	}

	if (month.length == 1) {
		month = "0" + month;
	}
	if (day.length == 1) {
		day = "0" + day;
	}
	if (year.length == 2) {
		year = "20" + year;
	}

	var digitsRE=/^\d{0,4}$/;
	if(!month.trim().match(digitsRE) || !day.trim().match(digitsRE) || !year.trim().match(digitsRE))return "";

	return month+"/"+day+"/"+year;
}

//Checks that an input string is an integer, with an optional +/- sign character.
function isInteger (s) {
	if (/^\s*(\+|-)?\d+\s*$/.test(s)) {
    	return true;
    }else{
    	return false;
    }
}
//Checks that an input string is a decimal number, with an optional +/- sign character.
function isDecimal (s) {
	if (/^\s*(\+|-)?((\d+(\.\d+)?)|(\.\d+))\s*$/.test(s)) {
    	return true;
    }else{
    	return false;
    }
}
//Check if string is currency
function isCurrency (s) {
	if (/^\s*(\+|-)?((\d+(\.\d\d)?)|(\.\d\d))\s*$/.test(s)) {
    	return true;
    }else{
    	return false;
    }
}
//Checks that an input string looks like a valid email address.
function isEmail (s) {
	if (/^\s*[\w\-\+_]+(\.[\w\-\+_]+)*\@[\w\-\+_]+\.[\w\-\+_]+(\.[\w\-\+_]+)*\s*$/.test(s)) {
    	return true;
    }else{
    	return false;
    }
}

	$.fn.showHTML = function(html, speed, callback) {
	      return this.each(function() {
	         // The element to be modified
	         var el = $(this);

	         // Preserve the original values of width and height - they'll need 
	         // to be modified during the animation, but can be restored once
	         // the animation has completed.
	         var finish = {width: this.style.width, height: this.style.height};

	         // The original width and height represented as pixel values.
	         // These will only be the same as `finish` if this element had its
	         // dimensions specified explicitly and in pixels. Of course, if that 
	         // was done then this entire routine is pointless, as the dimensions 
	         // won't change when the content is changed.
	         var cur = {width: el.width()+'px', height: el.height()+'px'};

	         // Modify the element's contents. Element will resize.
	         el.html(html);

	         // Capture the final dimensions of the element 
	         // (with initial style settings still in effect)
	         var next = {width: el.width()+'px', height: el.height()+'px'};

	         el .css(cur) // restore initial dimensions
	            .animate(next, speed, function()  // animate to final dimensions
	            {
	               el.css(finish); // restore initial style settings
	               if ( $.isFunction(callback) ) callback();
	            });
	      });
	};

function function_stringHasElements(datavalue,searchTags,mode){
	if(searchTags == null || searchTags.size == 0)
		return true;
	
	let retBool = true;
	if(mode == "OR")
		retBool = false;
	
	let datavalueSet = new Set();
	if(datavalue.includes(" ")){
		const datavalueArray = datavalue.split(" ");
		if(datavalueArray == null)
			return false;
		
		for (const dv of datavalueArray) { 
			datavalueSet.add(dv);
		}
	}else{
		datavalueSet.add(datavalue);
	}
	
	for(const searchTag of searchTags){
		if(mode == "OR"){
			retBool = (datavalueSet.has(searchTag)) || retBool;
		}else{
			retBool = (datavalueSet.has(searchTag)) && retBool;
		}
	}
	return retBool;
}
function function_createSWGTCircularProgressBar(elementName){
	var el = getElement(elementName); // get canvas

	var options = {
	    percent:  el.getAttribute('data-percent') || 25,
	    size: el.getAttribute('data-size') || 220,
	    lineWidth: el.getAttribute('data-line') || 15,
	    rotate: el.getAttribute('data-rotate') || 0,
		maxValue: el.getAttribute('data-maxvalue') || 100,
		showSizeOnly : el.getAttribute('data-showsizeonly') || 0,
		valueIsPercentage:  el.getAttribute('data-valueispercentage') || 0,
		valueColorClass: el.getAttribute('data-valuecolorclass') || "text-dark",
		donutBackgroundColor: el.getAttribute('data-donutbackgroundcolor') || "#efefef",
		donutValueColor: el.getAttribute('data-donutvaluecolor') || "#555555",
	}

	var canvas = document.createElement('canvas');
	var span = document.createElement('span');
	
	span.textContent = '';
	
	if(options.showSizeOnly == 1){
		span.textContent += options.percent;
	}else{
		span.textContent += options.percent + '/'+options.maxValue;
	}
	if(options.valueIsPercentage == 1)
		span.textContent += '%';
	

	if(options.valueIsPercentage == 1)
		span.className='pl-2 swgt-'+options.size+'-circle-progress-bar-span '+options.valueColorClass;
	else
		span.className='swgt-'+options.size+'-circle-progress-bar-span '+options.valueColorClass;
	    
	if (typeof(G_vmlCanvasManager) !== 'undefined') {
	    G_vmlCanvasManager.initElement(canvas);
	}

	var ctx = canvas.getContext('2d');
	canvas.width = canvas.height = options.size;
	canvas.className='swgt-circle-progress-bar-canvas';

	el.appendChild(span);
	el.appendChild(canvas);

	ctx.translate(options.size / 2, options.size / 2); // change center
	ctx.rotate((-1 / 2 + options.rotate / 180) * Math.PI); // rotate -90 deg

	var radius = (options.size - options.lineWidth) / 2;

	var drawCircle = function(color, lineWidth, percent) {
		percent = Math.min(Math.max(0, percent), 1);
		ctx.beginPath();
		ctx.arc(0, 0, radius, 0, Math.PI * 2 * percent, false);
		ctx.strokeStyle = color;
		ctx.lineCap = 'butt';
		ctx.lineWidth = lineWidth
		ctx.stroke();
	};

	drawCircle(options.donutBackgroundColor, options.lineWidth, 100 / 100);
	drawCircle(options.donutValueColor, options.lineWidth, options.percent / options.maxValue);
}

function function_formatTime(time,infinityText) {
	if(time == 0 || time == infinityText || time > 900000)
		return "&#8734;";
	
    // Hours, minutes and seconds
    var hrs = ~~(time / 3600);
    var mins = ~~((time % 3600) / 60);
    var secs = ~~time % 60;

    // Output like "1:01" or "4:03:59" or "123:03:59"
    var ret = "";
    if (hrs > 0) {
        if(hrs < 10)
        	ret += "0"+hrs+"h ";
        else
        	ret += ""+hrs+"h ";
    }else{
    	ret += "00h ";
    }
    if (mins > 0) {
        if(mins < 10)
        	ret += "0"+mins+"m ";
        else
        	ret += ""+mins+"m ";
    }else{
    	ret += "00m ";
    }
    if (secs > 0) {
        if(secs < 10)
        	ret += "0"+secs+"s";
        else
        	ret += ""+secs+"s";
    }else{
    	ret += "00s";
    }
    return ret;
}

        }

        if (!originalTableId) {
            $floatingTable.find('thead th').each(function () {
                const ariaControls = $(this).attr('aria-controls');
                if (ariaControls) {
                    originalTableId = ariaControls;
                    return false; // break out of .each()
                }
            });
        }

        if (!originalTableId) return;

        const $originalTable = $('#' + originalTableId);
        if (!$originalTable.length) return;

        const $sourceDt = $originalTable.closest('.swgt-dt');
        if (!$sourceDt.length) return;

        const isDense =
            $sourceDt.hasClass('swgt-dt--dense') ||
            $originalTable.hasClass('swgt-dt--dense') ||
            $originalTable.closest('.swgt-dt--dense').length > 0;

        $floatingTable.addClass('swgt-dt');

        if (isDense) {
            $floatingTable.addClass('swgt-dt--dense');
        } else {
            $floatingTable.removeClass('swgt-dt--dense');
        }
    });
}
var _swgtLazyLoadObserver = null;

function initSWGTLazyLoader() {
  if (_swgtLazyLoadObserver) return;

  if (!("IntersectionObserver" in window)) {
    $(".swgt-lazyload[data-src]").each(function () {
      var ds = this.getAttribute("data-src");
      if (ds) {
        this.src = ds;
        this.removeAttribute("data-src");
        this.classList.add("is-loaded");
      }
    });
    return;
  }

  _swgtLazyLoadObserver = new IntersectionObserver(function (entries) {
    entries.forEach(function (entry) {
      if (!entry.isIntersecting) return;

      var img = entry.target;
      var ds = img.getAttribute("data-src");

      if (ds) {
        img.onload = function () {
          img.classList.add("is-loaded");
          img.onload = null;
          img.onerror = null;
        };

        img.onerror = function () {
          img.classList.add("is-error");
          img.onload = null;
          img.onerror = null;
        };

        img.src = ds;
        img.removeAttribute("data-src");

        if (img.complete) {
          img.classList.add("is-loaded");
        }
      }

      img.removeAttribute("data-lazy-observed");
      _swgtLazyLoadObserver.unobserve(img);
    });
  }, {
    root: null,
    rootMargin: "200px 0px",
    threshold: 0.01
  });
}

function observeSWGTLazyLoad(container) {
  initSWGTLazyLoader();

  var $targets = container
    ? $(container).find(".swgt-lazyload[data-src]")
    : $(".swgt-lazyload[data-src]");

  if (_swgtLazyLoadObserver) {
    $targets.each(function () {
      if (this.dataset.lazyObserved === "true") return;
      this.dataset.lazyObserved = "true";
      _swgtLazyLoadObserver.observe(this);
    });
  }
}

function swgtFixDataTablesLayout() {
	if (!$.fn.DataTable || !$.fn.dataTable) {
		return;
	}

	try {
		var tables = $($.fn.dataTable.tables()).DataTable();

		tables.columns.adjust();

		if ($.fn.dataTable.Responsive) {
			tables.responsive.recalc();
		}

		if ($.fn.dataTable.FixedHeader) {
			setTimeout(function () {
				tables.fixedHeader.adjust();
			}, 50);
		}
	} catch (error) {
		if (window.console) {
			console.warn('SWGT DataTables layout adjustment failed:', error);
		}
	}
}

		var url = '/controllers/accountAnalytics/ajaxViewRuneRankAnalysis?selectedFocus=' + selectedFocus + '&runeID=' + runeID + '&nocache=' + noCache;

		if(classV)
			url += "&classV=" + classV;
		if(set_id)
			url += "&set_id=" + set_id;
		if(pri_eff)
			url += "&pri_eff=" + pri_eff;
		if(prefix_eff)

		if(sec_eff_4)
			url += "&sec_eff_4=" + sec_eff_4;
		if(extra)
			url += "&extra=" + extra;
		if(sec_eff_order)
			url += "&sec_eff_order=" + sec_eff_order;

		return url;
	}

	function getLoadingHtml() {
		var loadingText = window.SWGT_CONFIG && window.SWGT_CONFIG.phrases && window.SWGT_CONFIG.phrases.loadingPleaseWait ? window.SWGT_CONFIG.phrases.loadingPleaseWait : 'Loading, please wait...';

		return '<div class="col-12 px-0"><div class="text-center">' + loadingText + '</div></div>';
	}

	var modalTitle = window.SWGT_CONFIG && window.SWGT_CONFIG.phrases && window.SWGT_CONFIG.phrases.runeRankAnalysis ? window.SWGT_CONFIG.phrases.runeRankAnalysis : 'Rune - Rank - Analysis';

		title: modalTitle,
		type: BootstrapDialog.TYPE_PRIMARY,
		size: BootstrapDialog.SIZE_NORMAL,
		message: $('<div class="swgt-rune-rank-analysis-modal-body"></div>').html(getLoadingHtml()).load(url),
		closable: true,
		closeByBackdrop: true,
		closeByKeyboard: true,
		onshow: function(dialog) {
			document.querySelector('meta[name="viewport"]').content = 'width=device-width, minimum-scale=1.0, maximum-scale=1.0, initial-scale=1.0';
		},
		onshown: function(dialog){
			$(".modal-body").attr('style','padding: 0.5rem 1rem !important;');
			$(".modal-dialog").attr('style','margin-top:0.5rem !important;');

			var $modal = dialog.getModal();
			var $header = $modal.find('.bootstrap-dialog-header');
			var $closeButton = $header.find('.bootstrap-dialog-close-button');

			if(!$header.find('.swgt-rune-rank-refresh-btn').length){

					'<button type="button" class="swgt-rune-rank-refresh-btn" ' +

					});

						}

						$message.unblock();
						$button.removeClass('swgt-rune-rank-refresh-btn-loading');
					});
				});

				if($closeButton.length){
					$refreshButton.insertBefore($closeButton);
				}else{
					$header.append($refreshButton);
				}
			}
		},
		onhide: function(dialog) {
			document.querySelector('meta[name="viewport"]').content = 'width=device-width, initial-scale=1';
		}
	});

function function_showToastMessage(type, message) {
	type = type || "info";
	message = message || "";

	var allowedTypes = ["success", "warning", "danger", "info"];
	if(allowedTypes.indexOf(type) === -1)
		type = "info";

	var $container = $(".swgt-toast-container");
	if($container.length === 0) {
		$container = $("<div/>", {
			"class": "swgt-toast-container",
			"aria-live": "polite",
			"aria-atomic": "true"
		});
		$("body").append($container);
	}

	var $toast = $("<div/>", {
		"class": "swgt-toast-message swgt-toast-" + type,
		"text": message
	});

	$container.append($toast);

	setTimeout(function() {
		$toast.addClass("swgt-toast-show");
	}, 20);

	setTimeout(function() {
		$toast.removeClass("swgt-toast-show");

		setTimeout(function() {
			$toast.remove();
		}, 220);
	}, 4200);
}

var SWGT_SWIFT_SPEED_SET_BONUS = 0.25; //25%

function function_getSwiftSpeedSetBonus(baseSpeed){
