<script type="text/javascript">
function function_confirm(url,titleText,messageText){
	BootstrapDialog.show({
		animate: false,
		title: titleText,
        message: messageText,
        onshow: function(dialog) {
        	document.querySelector('meta[name="viewport"]').content = 'width=device-width, minimum-scale=1.0, maximum-scale=1.0, initial-scale=1.0';
        },
        onhide: function(dialog) {
        	document.querySelector('meta[name="viewport"]').content = 'width=device-width, initial-scale=1';
        },
		buttons: [
            {
            	label: 'Cancel',
            	cssClass: 'btn-sm btn-systemDefault mr-1',
            	action: function(dialogItself){
                	dialogItself.close();
            	}
        	}, 
        	{
            	label: 'Confirm',
            	cssClass: 'btn-sm btn-primary',
            	action: function(dialogItself){
            		dialogItself.close();
            		
					$.blockUI({ 
						message: '<h5>Processing, Please wait...<\/h5>', 
						css: {
							border: 'none',
							padding: '15px',
							backgroundColor: '#000', 
							'-webkit-border-radius': '10px', 
							'-moz-border-radius': '10px', 
							opacity: .5, 
							color: '#fff'
						} 
					});	
	
            		window.location=url;
            	}
        	}
        ]
    });
}
</script>
