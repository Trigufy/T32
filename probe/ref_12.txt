function function_import(){
	if(importAttachmentPond == null){
		function_common_modal_alert('Please select a JSON file to import');
		return;
	}

	const pondFile = importAttachmentPond.getFile();
	if(pondFile == null || pondFile.file == null){
		function_common_modal_alert('Please select a JSON file to import');
		return;
	}

	const file = pondFile.file;

	try {
		let reader = new FileReader();
		reader.onload = function (e) {
			try {
		    	let parsedJSON = JSON.parse(e.target.result);
				parsedJSON = parsedJSON[0]; //Take first
			
				let version = parsedJSON.version;
				let map_version = parsedJSON.map_version;
				let source = parsedJSON.source;
			
				
				let compatible = (map_version == '2025-03-23' && source == 'siegeSimulator') ? true : false;
				
			
				if(compatible){
					try{
						siegeMapStatsData = parsedJSON.siegeMapStatsData;
					}catch(err2) {}
					siegeMapData = parsedJSON.siege_map_data;

					function_loadMap();
					function_closeAll();
				}else{
					function_common_modal_alert('Incompatible simulator file');
				}
			}catch(err) {
				console.log(err.message);
				function_common_modal_alert('Unable to read simulator file');
			}
		};
		reader.onerror = function () {
			function_common_modal_alert('Unable to read simulator file');
		};
		reader.readAsText(file);
	}catch(err) {
		console.log(err.message);
		function_common_modal_alert('Unable to read simulator file');
	}
}

var analyzeTowersResult = {};
var analyzeTowersVisitedByColor = {};

function function_analyzeTowers() {
	analyzeTowersResult = {};
	analyzeTowersVisitedByColor = {};
	
    // initialize color tracking
    siegeMapData.forEach(({ color }) => {
		analyzeTowersResult[color] = { total: 0, connectedToHQ: 0, connectedTowerIDs: new Set() };
    });

    // count total towers per color
    siegeMapData.forEach(({ color }) => analyzeTowersResult[color].total++);

    // find HQs
    const hqs = siegeMapData.filter(({ isHQ }) => isHQ);
    
    // traverse from each HQ
    hqs.forEach(({ id, color }) => {
        if (!analyzeTowersVisitedByColor[color]) analyzeTowersVisitedByColor[color] = new Set();
        function_traverseTower(id, color);
    });

    // convert sets to arrays and count
    Object.keys(analyzeTowersResult).forEach(color => {
        analyzeTowersResult[color].connectedTowerIDs = Array.from(analyzeTowersResult[color].connectedTowerIDs);
        analyzeTowersResult[color].connectedToHQ = analyzeTowersResult[color].connectedTowerIDs.length;
    });
}

function function_traverseTower(towerId, color) {
    if (analyzeTowersVisitedByColor[color].has(towerId)) return;
    analyzeTowersVisitedByColor[color].add(towerId);

    const tower = siegeMapData.find(({ id }) => id === towerId);
    if (!tower || tower.color !== color) return;
    
    //if (!tower.isHQ) analyzeTowersResult[color].connectedTowerIDs.add(towerId);

    analyzeTowersResult[color].connectedTowerIDs.add(towerId);

    tower.connections.forEach(connId => {
        function_traverseTower(connId, color);
    });
}

function function_changeTower(selectedID){
 	var selectedColor = $("#MODAL_guild").val();
 	var selectedIcon = $("#MODAL_style").val();
 	var notes = $("#MODAL_notes").val();
 	var hpPercentage = $("#MODAL_hpPercentage").val()*1;
 	
 	//Find the json packet
 	for (const json of siegeMapData) {
		if(json.id == selectedID){
			switch(selectedColor){
				case "Blue":
					json.color = "Blue";
					break;
				case "Red":
					json.color = "Red";
					break;
				
				case "Yellow":
					json.color = "Yellow";
					break;
				
			}
			
			//Clear
			json.pingNumber = 0;
			json.prohibit = false;
			
			//Set
			switch(selectedIcon){
				case "NONE":
					//Do Nothing
					break;
				case "T1":
					json.pingNumber = 1;
					break;
				case "T2":
					json.pingNumber = 2;
					break;
				case "T3":
					json.pingNumber = 3;
					break;
				case "T4":
					json.pingNumber = 4;
					break;
				case "T5":
					json.pingNumber = 5;
					break;
				case "P":
					json.prohibit = true;
					break;
			}
			
			if(notes)
				json.memo = notes.replaceAll("\"",""); //double quote is not allowed
			else
				json.memo = "";
			
			if(hpPercentage > 0)
				json.hpPercentage = hpPercentage;
			else
				json.hpPercentage = 100;
			
			break;
		}
	}
 	function_closeAll();
	function_loadMap();
}
