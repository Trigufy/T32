	var typingTimers = {};
	
	// --- 1. Validate & clamp on typing/paste ---
	$(document).on("input", ".number-input", function () {

	    var $input = $(this);

	    var rawId = $input.attr("id");
	    var id = rawId.replace("artifactLineValue-", "");
	    
	    clearTimeout(typingTimers[id]);

	    typingTimers[id] = setTimeout(function () {
	    	var isDecimalAllowed = rawId.endsWith("-218");

		    var value = $input.val();

		    // Clean invalid characters
		    if (isDecimalAllowed) {
		        value = value
		            .replace(/[^0-9.]/g, '')
		            .replace(/(\..*?)\..*/g, '$1'); // only one decimal
		    } else {
		        value = value.replace(/[^0-9]/g, '');
		    }

		    if (value === "") {
		        $input.val("");
		        return;
		    }

		    var num = isDecimalAllowed ? parseFloat(value) : parseInt(value, 10);

		    if (isNaN(num)) {
		        $input.val("");
		        return;
		    }

		    // Clamp using artifactLineData
		    var key = "artifactSubStat-" + id;
		    if (artifactLineData[key]) {
		        var min = artifactLineData[key].min;
		        var max = artifactLineData[key].max;

		        if (num < min) num = min;
		        if (num > max) num = max;
		    }

		    // Fix decimal precision
		    if (isDecimalAllowed) num = num.toFixed(1);

		    $input.val(num);
		    
		    selectedOptionsValues.set((id*1),num);
	    }, 500); // 500ms delay after last keystroke
	});

	// --- 2. Arrow key increment/decrement ---
	$(document).on("keydown", ".number-input", function (e) {

	    if (e.key !== "ArrowUp" && e.key !== "ArrowDown") return;

	    e.preventDefault(); // stop cursor move / page scroll

	    var $input = $(this);

	    var rawId = $input.attr("id");
	    var id = rawId.replace("artifactLineValue-", "");

	    var isDecimalAllowed = rawId.endsWith("-218");
	    
	    var step = isDecimalAllowed ? 0.1 : 1;
	    if (e.key === "ArrowDown") step *= -1;

	    var value = $input.val();
	    var num = isDecimalAllowed ? parseFloat(value) : parseInt(value, 10);
	    if (isNaN(num)) num = 0;

	    num += step;

	    // Clamp using artifactLineData
	    var key = "artifactSubStat-" + id;
	    if (artifactLineData[key]) {
	        var min = artifactLineData[key].min;
	        var max = artifactLineData[key].max;

	        if (num < min) num = min;
	        if (num > max) num = max;
	    }

	    if (isDecimalAllowed) num = num.toFixed(1);

	    $input.val(num);
	    
	    selectedOptionsValues.set((id*1),num);
	});

});

function function_changeArtifactAttribute(button){
	artifactAttribute = $(button).data('artifact-attribute');
	artifactSide = $(button).data('artifact-side');
	
	//Clear all
	$('.swgt-artifact-attribute').each(function() {
		$(this).removeClass("btn-primary");
		$(this).addClass("btn-outline-primary");
		$(this).addClass("swgt-btn-transparent");
	});
	
	//Add the one we need
	$('.swgt-artifact-attribute').each(function() {
		if($(this).data('artifact-attribute') == artifactAttribute){
			$(this).removeClass("btn-outline-primary");
			$(this).removeClass("swgt-btn-transparent");
			$(this).addClass("btn-primary");
		}
	});
	
	function_updateArtifact();
}
function function_changeBuff(button,attribute){

			$('.swgt-'+attribute+'-buffed').removeClass("swgt-btn-transparent");

function function_changeSubStat(button){
	var substat = $(button).data('substat');
	if($(button).hasClass("btn-primary")){
		//Remove
		selectedOptions.delete(substat);
		selectedOptionsValues.delete(substat);
		$(button).removeClass("btn-primary");
		$(button).addClass("btn-outline-primary");
		$(button).addClass("swgt-btn-transparent");
	}else{
		//Add
		selectedOptions.set(substat,artifactLineData["artifactSubStat-"+substat].name);
		selectedOptionsValues.set(substat,0);
		$(button).removeClass("btn-outline-primary");
		$(button).removeClass("swgt-btn-transparent");
		$(button).addClass("btn-primary");
	}
	function_updateArtifact();
}

			$(this).addClass("btn-outline-primary");
			$(this).addClass("swgt-btn-transparent");
		}
	});
	selectedOptions = new Map();
	selectedOptionsValues = new Map();
	function_updateArtifact()
}
function function_updateArtifact(){
	var artifactBackground = "";
	var artifactIcon = "";
	var baseCDN = "https://do9d4mpqk497d.cloudfront.net";
	switch(selectedOptions.size) {
		case 1:
			artifactQuality = "Magic"
	    	break;
		case 2:
			artifactQuality = "Rare"
	    	break;
		case 3:
			artifactQuality = "Hero"
	    	break;
		case 4:
			artifactQuality = "Legend"
	    	break;
	  	default:
			artifactQuality = "Common"
	} 
	
	if("Water" == artifactAttribute || "Fire" == artifactAttribute || "Wind" == artifactAttribute || "Light" == artifactAttribute || "Dark" == artifactAttribute){
		artifactBackground = "element_"+artifactQuality.toLowerCase()+"_v2.png";
		artifactIcon = artifactAttribute.toLowerCase()+"_"+artifactQuality.toLowerCase()+"_v2.png";
	}else{
		artifactBackground = "archetype_"+artifactQuality.toLowerCase()+"_v2.png";
		artifactIcon = artifactAttribute.toLowerCase()+"_"+artifactQuality.toLowerCase()+"_v2.png";
	}
	//Background
	$(".artifactPreview").removeClass("artifact-quality-border-common");
	$(".artifactPreview").removeClass("artifact-quality-border-magic");
	$(".artifactPreview").removeClass("artifact-quality-border-rare");
	$(".artifactPreview").removeClass("artifact-quality-border-hero");
	$(".artifactPreview").removeClass("artifact-quality-background-legend");
	
	$(".artifactPreview").addClass("artifact-quality-border-"+artifactQuality.toLowerCase());

	//Icon
	$("#TD_artifactIcon").html(
   		"<div class=\"artifact-v2 rounded\" style=\"background-image:url("+baseCDN+"/common/images/artifact_icons/bg_"+artifactQuality.toLowerCase()+".png)\">"+
		"	<div class=\"artifact-style-v2 element\" style=\"background-image:url("+baseCDN+"/common/images/artifact_icons/"+artifactBackground+")\">"+
		"		<div class=\"artifact-symbol-v2\" style=\"background-image:url("+baseCDN+"/common/images/artifact_icons/"+artifactIcon+")\"></div>"+
    	"	</div>"+
    	"</div>"
    );
	
	//Title
	$("#artifactTitle").removeClass("artifact-quality-text-common");
	$("#artifactTitle").removeClass("artifact-quality-text-magic");
	$("#artifactTitle").removeClass("artifact-quality-text-rare");
	$("#artifactTitle").removeClass("artifact-quality-text-hero");
