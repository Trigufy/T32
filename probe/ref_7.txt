<script type="text/javascript">
function function_runReport(){
	var selectedElement = $("#dungeonKey :selected");
	
	if(selectedElement){
		let type = selectedElement.data('type');
		let dungeon_id = selectedElement.data('dungeonid');
		if(type == 'DUNGEON'){
			let stage_id = selectedElement.data('stageid');
			
			$.blockUI({ 
				message: '<h5>Loading Please Wait...<\/h5>',
				centerX: true,
				centerY: true,
				css: {
					border: 'none',
					padding: '15px',
					backgroundColor: '#000', 
					'-webkit-border-radius': '10px', 
					'-moz-border-radius': '10px', 
					opacity: .5, 
					color: '#fff'
				} 
			});
			
			$.ajax({
				type: "POST",
				url: "/controllers/dungeonAnalysis/ajaxLoadDungeon",
				data: {
					'dungeon_id': dungeon_id,
					'stage_id': stage_id,
					'startDate': $("#startDate").val(),
					'endDate': $("#endDate").val(),
					'renderOnSelectedTab': ''
				},
				success: function(html) {
					$("#DIV_reportContents").html(html);
					$.unblockUI();
				},
				error: function(html){
					function_setReportContentsError();
					$.unblockUI();
				}
			});
			
		}else if(type == 'RIFT'){
			let rift_dungeon_box_id = selectedElement.data('riftdungeonboxid');
			
			$.blockUI({ 
				message: '<h5>Loading Please Wait...<\/h5>',
				centerX: true,
				centerY: true,
				css: {
					border: 'none',
					padding: '15px',
					backgroundColor: '#000', 
					'-webkit-border-radius': '10px', 
					'-moz-border-radius': '10px', 
					opacity: .5, 
					color: '#fff'
				} 
			});
			
			$.ajax({
				type: "POST",
				url: "/controllers/dungeonAnalysis/ajaxLoadRaid",
				data: {
					'dungeon_id': dungeon_id,
					'rift_dungeon_box_id': rift_dungeon_box_id,
					'startDate': $("#startDate").val(),
					'endDate': $("#endDate").val(),
					'renderOnSelectedTab': ''
				},
				success: function(html) {
					$("#DIV_reportContents").html(html);
					$.unblockUI();
				},
				error: function(html){
					function_setReportContentsError();
					$.unblockUI();
				}
			});
			
		}else{
			function_setReportContentsError();
		}
	}else{
		function_setReportContentsError();
	}
}
function function_setReportContentsError(){
	$("#DIV_reportContents").html("<div class='alert alert-danger col-12 text-center' role='alert' style='margin-bottom:0px;'>An unexpected error has occurred</div>");
}
</script>

        	document.querySelector('meta[name="viewport"]').content = 'width=device-width, initial-scale=1';

            	cssClass: 'btn-sm btn-systemDefault',
